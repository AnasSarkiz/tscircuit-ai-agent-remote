import { withWinterSpec } from "lib/with-winter-spec"
import { z } from "zod"
import * as ZT from "lib/zod"
import { publicMapPackageBuild } from "lib/public-mapping/public-map-package-build"

const routeSpec = {
  methods: ["GET", "POST"],
  auth: "none",
  commonParams: z.object({
    package_build_id: z.string().uuid(),
    include_logs: z.boolean().optional().default(false),
  }),
  jsonResponse: z.object({
    ok: z.boolean(),
    package_build: ZT.public_package_build,
  }),
} as const

export default withWinterSpec(routeSpec)(async (req, ctx) => {
  const { package_build_id, include_logs } = req.commonParams

  const packageBuild = await ctx.unsafe_db
    .selectFrom("main.package_build as pb")
    .selectAll()
    .where("pb.package_build_id", "=", package_build_id)
    .executeTakeFirst()

  if (!packageBuild) {
    return ctx.error(404, {
      error_code: "package_build_not_found",
      message: "Package build not found",
    })
  }

  return ctx.json({
    ok: true,
    package_build: publicMapPackageBuild(packageBuild, {
      include_logs,
    }),
  })
})
