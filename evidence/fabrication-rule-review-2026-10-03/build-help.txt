Usage: tsci build [options] [file]

Run tscircuit eval and output circuit json

Arguments:
  file                             Path to the entry file

Options:
  --ci                             Run install and optional prebuild/build
                                   commands (or default CI build)
  --ignore-errors                  Do not exit with code 1 on errors
  --ignore-warnings                Do not log warnings
  --ignore-netlist-drc             Ignore netlist DRC errors/warnings
  --ignore-pin-specification-drc   Ignore pin-specification DRC errors/warnings
  --ignore-placement-drc           Ignore placement DRC errors/warnings
  --ignore-routing-drc             Ignore routing DRC errors/warnings
  --ignore-config                  Ignore options from tscircuit.config.json
  --disable-pcb                    Disable PCB outputs
  --routing-disabled               Disable routing during circuit generation
  --disable-parts-engine           Disable the parts engine
  --site                           Generate a static site in the dist directory
  --transpile                      Transpile the entry file to JavaScript
  --preview-images                 Generate preview images in the dist directory
  --all-images                     Generate preview images for every successful
                                   build output
  --pngs                           Generate PNG outputs during build generation
  --pcb-png                        Generate PCB PNG outputs during build
                                   generation
  --schematic-png                  Generate schematic PNG outputs during build
                                   generation
  --svgs                           Generate SVG outputs during build generation
  --pcb-svgs                       Generate PCB SVG outputs during build
                                   generation
  --schematic-svgs                 Generate schematic SVG outputs during build
                                   generation
  --simulation-svgs                Generate simulation graph SVG outputs during
                                   build generation
  --simulation-schematic-svgs      Generate schematic simulation SVG outputs
                                   during build generation
  --3d-png                         Generate 3D PNG outputs during build
                                   generation
  --3d                             Generate 3D PNG outputs during build
                                   generation
  --pcb-only                       Generate only PCB SVG outputs during build
                                   generation
  --schematic-only                 Generate only schematic SVG outputs during
                                   build generation
  --kicad-project                  Generate KiCad project directories for each
                                   successful build output
  --kicad-project-zip              Generate a zipped KiCad project for each
                                   successful build output
  --kicad-library                  Generate KiCad library in dist/kicad-library
  --kicad-library-name <name>      Specify the name of the KiCad library
  --preview-gltf                   Generate a GLTF file from the preview
                                   entrypoint
  --show-courtyards                Show courtyard outlines in PCB SVG outputs
  --glbs                           Generate GLB 3D model files for every
                                   successful build
  --step                           Generate STEP 3D model files for every
                                   successful build
  --profile                        Log per-circuit circuit.json generation time
                                   during build
  --autorouter-debug               Log autorouting phases and write
                                   unrouted/per-phase PNG artifacts
  --autorouter-phase <name>        Enable autorouting debugging through the
                                   named phase
  --autorouter-timeout <duration>  Abort an autorouting phase after a duration,
                                   e.g. "120s" or "2m"
  --autorouter-debug-dir <path>    Directory for autorouting debug artifacts
                                   (default: dist/autorouter-debug)
  --autorouter-dump-srj [mode]     Dump SimpleRouteJson inputs/outputs: "all",
                                   "failed", or "phase:<index>" (default:
                                   failed)
  --solver-debug                   Write constructor inputs for every solver
                                   event emitted during the build
  --solver-debug-dir <path>        Directory for solver input artifacts
                                   (default: dist/solver-debug)
  --kicad-pcm                      Generate KiCad PCM (Plugin and Content
                                   Manager) assets in dist/pcm
  --use-cdn-javascript             Use CDN-hosted JavaScript instead of bundled
                                   standalone file for --site
  --concurrency <number>           Number of files to build in parallel
                                   (default: 1) (default: "1")
  --inject-props <json>            Inject JSON props into the built file's
                                   default export
  --inject-props-file <path>       Inject JSON props from a file into the built
                                   file's default export
  -h, --help                       display help for command
